cmake_minimum_required(VERSION 3.0.2)project(airsim_drone_avoid)find_package(catkin REQUIRED COMPONENTS  rospy  std_msgs)catkin_package(  CATKIN_DEPENDS rospy std_msgs)include_directories(  ${catkin_INCLUDE_DIRS})install(PROGRAMS  src/main.py  src/model.py  src/airsim_utils.py  src/inference.py  src/train_avoid_model.py  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})install(DIRECTORY launch/  DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/launch)
